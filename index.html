/** * 75 HARD TRACKER - GOOGLE APPS SCRIPT WEB APP * * Setup Instructions: * 1. Open Google Sheets, name Sheet1 "75 Hard Log". * 2. Go to Extensions > Apps Script, paste this entire file, and click Save (floppy disk). * 3. Click Deploy > Manage Deployments > Edit (pencil) > Version: "New version" > Deploy. */ function setupHeaders() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var headers = [ "Timestamp", "Day #", "Date", "Weight (lbs)", "Water (128oz)", "Outdoor Workout", "Workout 2", "Book Title", "Pages Read", "Strict Diet", "No Alcohol", "Photo Done", "Notes" ]; if (sheet.getLastRow() === 0) { sheet.appendRow(headers); var headerRange = sheet.getRange(1, 1, 1, headers.length); headerRange.setFontWeight("bold"); headerRange.setBackground("#18181b"); headerRange.setFontColor("#f4f4f5"); } } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { setupHeaders(); } var data = {}; if (e.postData && e.postData.contents) { data = JSON.parse(e.postData.contents); } else if (e.parameter) { data = e.parameter; } var timestamp = new Date(); var dayNumber = data.dayNumber || "N/A"; var dateString = data.date || Utilities.formatDate(timestamp, Session.getScriptTimeZone(), "yyyy-MM-dd"); var weight = data.weight ? (data.weight + " lbs") : "N/A"; var waterComplete = (data.water1 && data.water2) ? "YES (128oz)" : "INCOMPLETE"; var outdoorWorkout = data.workout1Complete ? ("YES - " + (data.workout1Desc || "Outdoor 45m")) : "NO"; var workout2 = data.workout2Complete ? ("YES - " + (data.workout2Desc || "Workout 2 45m")) : "NO"; var bookTitle = data.bookTitle || "N/A"; var pagesRead = data.pagesRead || "N/A"; var dietFollowed = data.dietFollowed ? "YES" : "NO"; var noAlcohol = data.noAlcohol ? "YES" : "NO"; var photoDone = data.photoTaken ? "YES" : "NO"; var notes = data.notes || ""; sheet.appendRow([ timestamp, dayNumber, dateString, weight, waterComplete, outdoorWorkout, workout2, bookTitle, pagesRead, dietFollowed, noAlcohol, photoDone, notes ]); return ContentService.createTextOutput(JSON.stringify({ "status": "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function doGet(e) { return ContentService.createTextOutput("75 Hard Webhook is active and listening for POST requests."); }
